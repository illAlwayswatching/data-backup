void RankViewUI::_createRebolt(){

    _rankViewNode = CompoRebolt::safeLoadWithSubNotify<Layer*>("排行榜_合集页.redream");

        if (notify == "点击Players按钮") {
            CCLOG("RRRRr1.2: 处理点击Players按钮");
            updatePlayersWorldList(false);
            updatePlayersRegionList(false);
        }
        else if (notify == "点击Teams按钮") {
            CCLOG("RRRRr1.3: 处理点击Teams按钮");
            updateTeamWorldList(false);
            updateTeamRegionList(false);

                CCLOG("RRRRr1.5: 设置_topNode成功");
            }
            if(param == "列表区域顶部裁切节点"){
                _topEdgeNode = node;
                CCLOG("RRRRr1.6: 设置_topEdgeNode成功");
            }

                CCLOG("RRRRr1.7: 设置_downNode和_downEdgeNode成功");
